-- ==================== KEY INPUT ====================

local actualKey = ""
local updatingKey = false

KeyBox:GetPropertyChangedSignal("Text"):Connect(function()
    if updatingKey then return end

    updatingKey = true

    local text = KeyBox.Text
    local len = utf8.len(text) or #text
    local actualLen = utf8.len(actualKey) or #actualKey

    if len > actualLen then
        local added = string.sub(text, #text, #text)
        actualKey = actualKey .. added
    elseif len < actualLen then
        local diff = actualLen - len
        actualKey = string.sub(actualKey, 1, actualLen - diff)
    end

    KeyBox.Text = string.rep("•", utf8.len(actualKey) or #actualKey)
    updatingKey = false
end)

-- ==================== FPS PROPERTY ====================

local function setTemporaryProperty(instance, propertyName, newValue)
    local ok, originalValue = pcall(function()
        return instance[propertyName]
    end)

    if not ok then return end

    if not originalProperties[instance] then
        originalProperties[instance] = {}
    end

    if originalProperties[instance][propertyName] == nil then
        originalProperties[instance][propertyName] = originalValue
    end

    pcall(function()
        instance[propertyName] = newValue
    end)
end

local function applyLowGraphics(instance)
    if not instance or not instance.Parent then return end

    if instance:IsA("BasePart") then
        setTemporaryProperty(instance, "Material", Enum.Material.SmoothPlastic)
        setTemporaryProperty(instance, "CastShadow", false)
        setTemporaryProperty(instance, "Reflectance", 0)

    elseif instance:IsA("Decal") or instance:IsA("Texture") then
        setTemporaryProperty(instance, "Transparency", 1)

    elseif instance:IsA("PostEffect") then
        setTemporaryProperty(instance, "Enabled", false)

    elseif instance:IsA("ParticleEmitter")
        or instance:IsA("Trail")
        or instance:IsA("Beam")
        or instance:IsA("Fire")
        or instance:IsA("Smoke")
        or instance:IsA("Sparkles")
        or instance:IsA("Clouds") then

        setTemporaryProperty(instance, "Enabled", false)
    end
end

local function restoreGraphics()
    for instance, properties in pairs(originalProperties) do
        for propertyName, originalValue in pairs(properties) do
            pcall(function()
                instance[propertyName] = originalValue
            end)
        end
    end

    table.clear(originalProperties)
end

local function setFpsBoost(enabled)
    fpsBoostEnabled = enabled

    if enabled then
        FpsBoostButton.Text = "FPS BOOST: ON"
        FpsBoostButton.BackgroundColor3 = Color3.fromRGB(25, 130, 65)

        setTemporaryProperty(Lighting, "GlobalShadows", false)
        setTemporaryProperty(Lighting, "EnvironmentDiffuseScale", 0)
        setTemporaryProperty(Lighting, "EnvironmentSpecularScale", 0)

        local terrain = workspace:FindFirstChildOfClass("Terrain")

        if terrain then
            setTemporaryProperty(terrain, "Decoration", false)
            setTemporaryProperty(terrain, "WaterWaveSize", 0)
            setTemporaryProperty(terrain, "WaterWaveSpeed", 0)
            setTemporaryProperty(terrain, "WaterReflectance", 0)
        end

        for _, inst in ipairs(workspace:GetDescendants()) do
            applyLowGraphics(inst)
        end

        for _, inst in ipairs(Lighting:GetDescendants()) do
            applyLowGraphics(inst)
        end

        if not workspaceAddedConnection then
            workspaceAddedConnection =
                workspace.DescendantAdded:Connect(applyLowGraphics)
        end

        if not lightingAddedConnection then
            lightingAddedConnection =
                Lighting.DescendantAdded:Connect(applyLowGraphics)
        end
    else
        FpsBoostButton.Text = "FPS BOOST: OFF"
        FpsBoostButton.BackgroundColor3 = Color3.fromRGB(45, 45, 55)

        if workspaceAddedConnection then
            workspaceAddedConnection:Disconnect()
            workspaceAddedConnection = nil
        end

        if lightingAddedConnection then
            lightingAddedConnection:Disconnect()
            lightingAddedConnection = nil
        end

        restoreGraphics()
    end
end

-- ==================== DETEKSI KEY ====================

local function isRealKeyPrompt(prompt)
    if not prompt:IsA("ProximityPrompt") then
        return false
    end

    local action = string.lower(prompt.ActionText or "")
    local object = string.lower(prompt.ObjectText or "")

    if action ~= "pick up" and action ~= "ambil" and action ~= "" then
        return false
    end

    if not string.find(object, "key", 1, true) and not string.find(object, "kunci", 1, true) then
        return false
    end

    if string.find(object, "padlock", 1, true)
        or string.find(object, "lock", 1, true)
        or string.find(object, "gembok", 1, true) then
        return false
    end

    return true
end

-- ==================== DETEKSI CODE ====================

local function containsCodeText(value)
    local text = string.lower(tostring(value or ""))

    return string.find(text, "catatan", 1, true)
        or string.find(text, "kode", 1, true)
        or string.find(text, "code", 1, true)
        or string.find(text, "sandi", 1, true)
end

local function isRealCodePrompt(prompt)
    if not prompt:IsA("ProximityPrompt") then
        return false
    end

    local action = string.lower(prompt.ActionText or "")
    local object = prompt.ObjectText or ""

    if not containsCodeText(object) then
        return false
    end

    if action ~= "" and action ~= "ambil" and action ~= "pick up" then
        return false
    end

    return true
end

local function isCodeObject(obj)
    local name = string.lower(obj.Name or "")

    return string.find(name, "catatan", 1, true)
        or string.find(name, "kode", 1, true)
        or string.find(name, "code", 1, true)
        or string.find(name, "sandi", 1, true)
end

-- ==================== GET OBJECT ====================

local function getPromptObject(prompt)
    local current = prompt.Parent

    while current and current ~= workspace do
        if current:IsA("Model") then
            return current
        end

        current = current.Parent
    end

    return prompt.Parent
end

-- ==================== TELEPORT SYSTEM ====================

local function teleportToObject(object)
    if not object or not object.Parent then
        return false
    end

    local character = player.Character
    local rootPart = character and character:FindFirstChild("HumanoidRootPart")

    if not rootPart then
        return false
    end

    local targetPart

    if object:IsA("BasePart") then
        targetPart = object
    else
        targetPart = object:FindFirstChildWhichIsA("BasePart", true)
    end

    if not targetPart then
        return false
    end

    rootPart.CFrame = CFrame.new(
        targetPart.Position + Vector3.new(0, 3, 0)
    )

    return true
end

local function findKeyObject()
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("ProximityPrompt") and isRealKeyPrompt(obj) then
            local keyObject = getPromptObject(obj)

            if keyObject and keyObject.Parent then
                return keyObject
            end
        end
    end

    return nil
end

local function findCodeObject()
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("ProximityPrompt") and isRealCodePrompt(obj) then
            local codeObject = getPromptObject(obj)

            if codeObject and codeObject.Parent then
                return codeObject
            end
        end
    end

    for _, obj in ipairs(workspace:GetDescendants()) do
        if (obj:IsA("Model") or obj:IsA("BasePart"))
            and isCodeObject(obj) then
            return obj
        end
    end

    return nil
end

local autoTeleportDebounce = false

local function autoTeleportAfterPickup()
    if autoTeleportDebounce then return end

    autoTeleportDebounce = true

    task.delay(0.5, function()
        local character = player.Character
        local rootPart = character
            and character:FindFirstChild("HumanoidRootPart")

        if rootPart then
            rootPart.CFrame = CFrame.new(DESTINATION)
        end

        task.delay(1, function()
            autoTeleportDebounce = false
        end)
    end)
end

ProximityPromptService.PromptTriggered:Connect(function(prompt, triggeringPlayer)
    if triggeringPlayer and triggeringPlayer ~= player then
        return
    end

    if isRealKeyPrompt(prompt) or isRealCodePrompt(prompt) then
        autoTeleportAfterPickup()
    end
end)

-- ==================== TAMBAHKAN ESP PLAYER ====================

local function addPlayerESP(p)
    if p == player then return end

    local function setupCharacter(char)
        if not char then return end
        if activePlayerHighlights[char] then return end

        local highlight = Instance.new("Highlight")
        highlight.Name = "ESP_Player_Highlight"
        highlight.Adornee = char
        highlight.FillColor = Color3.fromRGB(0, 255, 100)
        highlight.FillTransparency = 0.4
        highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
        highlight.OutlineTransparency = 0
        highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        highlight.Parent = char

        activePlayerHighlights[char] = highlight

        createESPLabel(
            char,
            p.Name,
            Color3.fromRGB(0, 255, 100),
            activePlayerBillboards
        )
    end

    if p.Character then
        setupCharacter(p.Character)
    end

    p.CharacterAdded:Connect(setupCharacter)
end

local function updatePlayerESP()
    if not espPlayerActive then return end

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player then
            local char = p.Character
            if char and char.Parent then
                if not activePlayerHighlights[char] then
                    addPlayerESP(p)
                end
            end
        end
    end

    for char, highlight in pairs(activePlayerHighlights) do
        if not char or not char.Parent then
            if highlight then highlight:Destroy() end
            activePlayerHighlights[char] = nil

            local billboard = activePlayerBillboards[char]
            if billboard then billboard:Destroy() end
            activePlayerBillboards[char] = nil
        end
    end
end

local function scanCodeObjects()
    if tick() - lastCodeScan < 0.5 then return end
    lastCodeScan = tick()

    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("ProximityPrompt") and isRealCodePrompt(obj) then
            addCodeESP(getPromptObject(obj))

        elseif (obj:IsA("Model") or obj:IsA("BasePart"))
            and isCodeObject(obj) then
            addCodeESP(obj)
        end
    end

    for obj, highlight in pairs(activeCodeHighlights) do
        if not obj.Parent then
            if highlight then highlight:Destroy() end
            activeCodeHighlights[obj] = nil

            local billboard = activeCodeBillboards[obj]
            if billboard then billboard:Destroy() end
            activeCodeBillboards[obj] = nil
        end
    end
end

-- ==================== UPDATE ESP OPAH ====================

local function updateOpahESP()
    if trackedOpahModel and trackedOpahModel.Parent then
        return
    end

    clearOpahESP()

    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Model") then
            local humanoid = obj:FindFirstChildOfClass("Humanoid")

            local hrp = obj:FindFirstChild("HumanoidRootPart")
                or obj:FindFirstChild("Torso")
                or obj:FindFirstChild("Head")

            local isAPlayer = false

            for _, p in ipairs(Players:GetPlayers()) do
                if p.Character == obj then
                    isAPlayer = true
                    break
                end
            end

            if not isAPlayer and humanoid and hrp then
                local isTallOpah = false

                pcall(function()
                    if obj:GetExtentsSize().Y > 4.3 then
                        isTallOpah = true
                    end
                end)

                if isTallOpah then
                    local rayOrigin = hrp.Position
                    local rayDirection = Vector3.new(0, 30, 0)

                    local raycastParams = RaycastParams.new()
                    raycastParams.FilterType =
                        Enum.RaycastFilterType.Exclude

                    raycastParams.FilterDescendantsInstances = {
                        obj,
                        player.Character
                    }

                    local raycastResult = workspace:Raycast(
                        rayOrigin,
                        rayDirection,
                        raycastParams
                    )

                    if raycastResult then
                        trackedOpahModel = obj

                        local highlight = Instance.new("Highlight")
                        highlight.Name = "OpahHighlight"
                        highlight.Adornee = obj
                        highlight.FillColor = Color3.fromRGB(255, 40, 40)
                        highlight.FillTransparency = 0.4
                        highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
                        highlight.OutlineTransparency = 0
                        highlight.Parent = obj

                        activeHighlight = highlight

                        local billboard = Instance.new("BillboardGui")
                        billboard.Name = "OpahNameTag"
                        billboard.Size = UDim2.new(0, 180, 0, 50)
                        billboard.StudsOffset = Vector3.new(0, 5, 0)
                        billboard.AlwaysOnTop = true
                        billboard.Parent = hrp

                        local nameLabel = Instance.new("TextLabel")
                        nameLabel.Size = UDim2.new(1, 0, 1, 0)
                        nameLabel.BackgroundTransparency = 1
                        nameLabel.TextColor3 = Color3.fromRGB(255, 50, 50)
                        nameLabel.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
                        nameLabel.TextStrokeTransparency = 0
                        nameLabel.TextSize = 22
                        nameLabel.Font = Enum.Font.GothamBold
                        nameLabel.Text = "WARNING OPAH"
                        nameLabel.Parent = billboard

                        activeBillboard = billboard
                        break
                    end
                end
            end
        end
    end
end

-- ==================== LOGO DRAG ====================

local isLogoDragging = false
local logoDragStartPos = Vector2.new()
local logoStartPos = UDim2.new()
local logoHasMoved = false

LogoButton.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        isLogoDragging = true
        logoDragStartPos = input.Position
        logoStartPos = LogoButton.Position
        logoHasMoved = false
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if isLogoDragging and (
        input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch
    ) then
        isLogoDragging = false

        if not logoHasMoved then
            toggleMenu()
        end
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if isLogoDragging and (
        input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch
    ) then

        local delta = input.Position - logoDragStartPos

        if math.abs(delta.X) > 4 or math.abs(delta.Y) > 4 then
            logoHasMoved = true
        end

        if logoHasMoved then
            LogoButton.Position = UDim2.new(
                logoStartPos.X.Scale,
                logoStartPos.X.Offset + delta.X,
                logoStartPos.Y.Scale,
                logoStartPos.Y.Offset + delta.Y
            )
        end
    end
end)

-- ==================== KEYBOARD SHORTCUT ====================

local teleportCycle = 1

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if input.KeyCode == Enum.KeyCode.L then
        if keyAuthenticated and (not KeyFrame or not KeyFrame.Parent) then
            toggleMenu()
        end
    end

    if gameProcessed then return end

    if input.KeyCode == Enum.KeyCode.H then
        if not keyAuthenticated then return end

        if UserInputService:GetFocusedTextBox() then
            return
        end

        if teleportCycle >= 1 and teleportCycle <= 5 then
            local keyObject = findKeyObject()

            if keyObject and teleportToObject(keyObject) then
                TeleportKeyButton.Text = "KEY " .. tostring(teleportCycle) .. " FOUND!"
            else
                TeleportKeyButton.Text = "KEY NOT FOUND!"
            end

            teleportCycle = teleportCycle + 1

            task.delay(2, function()
                if TeleportKeyButton.Parent then
                    TeleportKeyButton.Text = "TELEPORT TO KEY"
                end
            end)
        else
            local codeObject = findCodeObject()

            if codeObject and teleportToObject(codeObject) then
                TeleportCodeButton.Text = "CODE FOUND!"
            else
                TeleportCodeButton.Text = "CODE NOT FOUND!"
            end

            teleportCycle = 1

            task.delay(2, function()
                if TeleportCodeButton.Parent then
                    TeleportCodeButton.Text = "TELEPORT TO CODE"
                end
            end)
        end
    end
end)
